#!/usr/bin/env python3
"""Logos Pro printable 1 kg spool stand v0.1. PETG / Bambu A1 mini.

Four parts: one frame, two identical A-stands, and one D-section rod with end stops.
No bearings or screws. Prototype; physical fit/load test still required.
Units in mm. No printer movement / printing actions.
"""
from pathlib import Path
import cadquery as cq
import trimesh
import json, hashlib, math

OUT = Path('/mnt/data/logos_petg_spool_holder')
OUT.mkdir(parents=True, exist_ok=True)

# 1 kg Bambu-style spool: approx 200 mm max outer diameter, up to ~80 mm width.
# Strong PETG shell (recommended >=4 walls, 25%-35% gyroid).
STAND_X = 59.0
STAND_T = 12.0
BASE_L = 158.0
BASE_D = 136.0
BASE_H = 9.0
LEG_TONGUE_Y = 63.0
TONGUE_CLEARANCE = 0.35 # each side
AXLE_RADIUS = 12.5
SADDLE_RADIUS = 12.9
AXLE_CENTER_HEIGHT = 138.0


def box(x, y, z, dx, dy, dz):
    return cq.Workplane('XY').box(dx,dy,dz,centered=(True,True,False)).translate((x,y,z))

# Strong low-profile perimeter frame and central stabilizing rail; two sockets per part.
frame = box(-62.0,0,0,25,BASE_D,BASE_H).union(box(62.0,0,0,25,BASE_D,BASE_H))
for y in (-59.0,59.0):
    frame = frame.union(box(0,y,0,BASE_L,18.0,BASE_H))
frame = frame.union(box(0,0,0,135.0,13.0,BASE_H))
# tongue openings can be optionally glue-fit after successful dry fit.
for x in (-STAND_X,STAND_X):
    frame = frame.cut(box(x,0,-1,STAND_T+2*TONGUE_CLEARANCE,LEG_TONGUE_Y+2*TONGUE_CLEARANCE,BASE_H+2))
frame=frame.clean()

# A-frame stand, extruded along X. Each stays in original assembled world coordinates.
# CadQuery YZ: coordinates (Y, Z), normal +X.
def yzshape(points):
    return cq.Workplane('YZ').polyline(points).close().extrude(STAND_T/2, both=True)

# Foot rail, slanted left/right braces and an upper cradle block.
leg = yzshape([(-54,BASE_H), (54,BASE_H), (54,23), (-54,23)])
leg = leg.union(yzshape([(-55,16),(-39,16),(-8,131),(-26,131)]))
leg = leg.union(yzshape([(55,16),(39,16),(8,131),(26,131)]))
leg = leg.union(yzshape([(-27,117),(27,117),(27,151),(-27,151)]))
# Lower tongue: long rectangular mortise, tight but not force-fit.
leg = leg.union(yzshape([(-LEG_TONGUE_Y/2,0),(LEG_TONGUE_Y/2,0),(LEG_TONGUE_Y/2,BASE_H),(-LEG_TONGUE_Y/2,BASE_H)]))
# The round saddle and vertical escape path allow the rod to be lowered from above.
saddle = cq.Workplane('YZ').center(0, AXLE_CENTER_HEIGHT).circle(SADDLE_RADIUS).extrude(STAND_T/2+2,both=True)
leg = leg.cut(saddle).cut(box(0,0,AXLE_CENTER_HEIGHT, STAND_T+5,2*SADDLE_RADIUS,30))
leg = leg.clean()
# Original assembly position of the two identical stands
stand_left = leg.translate((-STAND_X,0,0))
stand_right = leg.translate((STAND_X,0,0))

# Shaft with end stops; D-shaped lower chord for support-free horizontal PETG print.
# Put shaft on print bed with broad flat; installed with same flat facing down.
axis = cq.Workplane('YZ').center(0,0).circle(AXLE_RADIUS).extrude(78, both=True)
# Both end guards sit outside the two stands, allowing them to keep axle captive.
for x in (-70.5,70.5):
    axis=axis.union(cq.Workplane('YZ').workplane(offset=x-1.6).circle(17).extrude(3.2))
# Cut common flat in lower semicircle (not through the spool's load-bearing top).
axis=axis.cut(box(0,0,-40,180,50,30.5)) # removes z < -9.5
axis=axis.clean()

# Export individual parts flat on the bed, correct CAD scaling. CadQuery solid models are millimeter.
# Frame is already flat on the XY plane, and shaft along X with bottom at 0.
# Leg needs to be turned from upright YZ into XY plane, with thickness vertical as printed.
# We rotate world around Y by +90 deg: X thickness becomes Z thickness.
stand_print = leg.rotate((0,0,0),(0,1,0),90)
# after rotation, min Z should be -T/2; set on bed and center in XY.
bbox = stand_print.val().BoundingBox()
stand_print = stand_print.translate((-(bbox.xmin+bbox.xmax)/2, -(bbox.ymin+bbox.ymax)/2, -bbox.zmin))
axis_print=axis.translate((0,0,9.5))

parts = {
    'base_frame':frame,
    'upright_A_print_twice': stand_print,
    'axle_D_section':axis_print,
}

checks = {}
for name, shape in parts.items():
    fp=OUT/(name+'.stl')
    cq.exporters.export(shape, str(fp), tolerance=0.10, angularTolerance=0.24)
    mesh=trimesh.load_mesh(fp,process=True)
    bb=mesh.bounds
    dims=(bb[1]-bb[0]).tolist()
    info={
      'file':fp.name,
      'sha256':hashlib.sha256(fp.read_bytes()).hexdigest(),
      'dimensions_mm':[round(float(x),2) for x in dims],
      'volume_cm3':round(mesh.volume/1000,2),
      'watertight':bool(mesh.is_watertight),
      'winding_consistent':bool(mesh.is_winding_consistent),
      'components':len(mesh.split(only_watertight=False)),
      'faces':int(len(mesh.faces)),
      'fits_A1_mini_build_area': max(dims[:2]) <= 180 and min(dims)>=0 and dims[2] <= 180,
      'on_bed':abs(float(bb[0][2])) < 0.2,
    }
    checks[name]=info
    print(name, json.dumps(info,ensure_ascii=False))

# Validate assemblies and clearances with known nominal spool envelope.
# Upright sockets deliberately allow 0.35 mm on each side for PETG tolerance.
assembly = [frame,stand_left,stand_right,axis.translate((0,0,AXLE_CENTER_HEIGHT))]
# Shaft guard at +/-70.5, each 3.2 thickness, clear of support outer surfaces +/-65.
# Printed axle section dimensions after flat cut <=158 x 34 x 26.5 mm.
summary={
  'name':'Logos Pro Desk Spool Stand v0.1',
  'target_printer':'Bambu Lab A1 mini',
  'material':'PETG (1 color, no AMS)',
  'max_nominal_spool':'1 kg, diameter 200 mm, width up to 85 mm, core opening >=36 mm',
  'parts_to_print':{'base_frame':1,'upright_A_print_twice':2,'axle_D_section':1},
  'no_extra_hardware':True,
  'assembly_length_mm':BASE_L,
  'assembly_depth_mm':BASE_D,
  'axis_height_mm':AXLE_CENTER_HEIGHT,
  'fit_tolerance_per_side_mm':TONGUE_CLEARANCE,
  'manufacturing':{'nozzle_mm':0.4,'layer_height_mm':0.2,'walls':4,'infill':'25-35% gyroid',
                   'supports':'none for base/stand; review slicer for axle','brim':'recommended 5mm for axle'},
  'status':'CAD prototype; require physical dry-fit and printer-specific slicing verification',
  'checks':checks,
}
(OUT/'manifest.json').write_text(json.dumps(summary,indent=2,ensure_ascii=False))
assert all(x['watertight'] and x['winding_consistent'] and x['components']==1 and x['fits_A1_mini_build_area'] and x['on_bed'] for x in checks.values()),'Validation failed'
print('CAD_VALIDATED',OUT)
